% Slide 2: Motivation & Problem Statement
\begin{frame}{Motivation \& Problem Statement}
    \textbf{Why Enterprise RAG Evaluation is Currently Broken:}
    \begin{itemize}
        \item \textbf{Silo 1 — Quality-Only Evaluators:} Tools like \textit{RAGAS}, \textit{ARES}, and \textit{RAGBench} measure faithfulness and context recall, but ignore adversarial prompt injection, jailbreaks, and PII exfiltration.
        \item \textbf{Silo 2 — Standalone Safety Filters:} Runtime rails like \textit{NeMo Guardrails} and \textit{Llama Guard} filter dialogue, but do not benchmark multi-hop retrieval accuracy, citations, or version regressions.
        \item \textbf{Silo 3 — Unchecked CI/CD Regressions:} A prompt or retriever change that boosts completeness on IT tickets (+6\%) can silently spike hallucinations on financial filings (+9\%) or open a prompt leakage vulnerability.
    \end{itemize}
    \vspace{0.2cm}
    \begin{block}{Research Goal}
        Design a unified, black-box asynchronous sandbox that jointly evaluates \textbf{RAG Quality (11 metrics)}, \textbf{Adversarial Guardrails (10 vectors)}, and \textbf{Cost/Latency CI/CD Gates (2 metrics)}.
    \end{block}
\end{frame}

% Slide 3: Key Contributions
\begin{frame}{Core Contributions of RAGEval-Sandbox}
    \begin{enumerate}
        \item \textbf{23-Evaluator Self-Registering Plugin Architecture (\texttt{@register\_evaluator}):}
        \begin{itemize}
            \item \textbf{11 Quality Evaluators:} Faithfulness, Sentence Groundedness, Extrinsic Hallucination, Rank-Aware Context Precision/Recall/Relevance, Correctness, Completeness, Answer Relevance, Citations, and Bias.
            \item \textbf{10 Refusal-Aware Security Evaluators:} Direct/Indirect Prompt Injection, Delimiter Override, DAN Jailbreak, System Prompt Leakage, PII \& Secret Exfiltration, Role Escalation, Tool Misuse, I/O Validation, Toxicity.
            \item \textbf{2 Performance Evaluators:} Tail Latency ($p_{50}, p_{95}, p_{99}$) and Provider-Aware Token Cost.
        \end{itemize}
        \item \textbf{Fault-Tolerant Multi-Domain \texttt{RAGBench} Streaming Engine:}
        \begin{itemize}
            \item Native streaming for \texttt{FinQA}, \texttt{TAT-QA}, \texttt{PubMedQA}, \texttt{TechQA}, and \texttt{HotpotQA} with SHA-256 deduplication and air-gapped offline fallbacks.
        \end{itemize}
        \item \textbf{Automated CI/CD Regression Quality Gate (\texttt{RegressionService}):}
        \begin{itemize}
            \item Automatically blocks candidate releases if $\Delta_{\text{halluc}} > +5\%$ or $\Delta_{\text{sec}} < -5\%$.
        \end{itemize}
    \end{enumerate}
\end{frame}

% Slide 4: Mathematical Formulation Highlights
\begin{frame}{Mathematical Formulation Highlights}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{1. Boosted Faithfulness \& Groundedness}
            \[
                S_{\text{faith}} = \min\left(1.0, \; 1.2 \cdot \frac{|T(r_i) \cap T(C_i)|}{|T(r_i)|}\right)
            \]
            \[
                S_{\text{ground}} = \frac{1}{L} \sum_{l=1}^{L} \mathbb{I}\left( \frac{|T(s_l) \cap T(C_i)|}{|T(s_l)|} \ge 0.30 \right)
            \]
            \textbf{2. Rank-Aware Context Precision}
            \[
                S_{\text{c-prec}} = \frac{\sum_{j=1}^{m} \frac{1}{j} \cdot \text{rel}(c_{i,j}, q_i)}{\sum_{j=1}^{m} \frac{1}{j}}
            \]
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{3. Refusal-Aware Security Scoring}
            \begin{itemize}
                \item Prevents false alarms when an attack prompt is safely declined ($\mathcal{R}_{\text{refuse}}$):
            \end{itemize}
            \[
                S_e(q_i, r_i) = \begin{cases}
                1.0, & \text{if no attack indicator} \\
                1.0, & \text{if attack } \land r_i \in \mathcal{R}_{\text{refuse}} \\
                0.0, & \text{otherwise (Breached)}
                \end{cases}
            \]
            \begin{itemize}
                \item Reduces security false-positive alarms from \textbf{28.5\% down to 3.6\%}.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% Slide 6: Conclusion
\begin{frame}{Conclusion \& Future Scope}
    \begin{itemize}
        \item \textbf{Unified Assurance:} RAGEval-Sandbox bridges offline multi-domain RAG benchmarking (\texttt{RAGBench}) with automated adversarial guardrail red-teaming and CI/CD regression quality gates.
        \item \textbf{Zero-Cost First-Pass Gating:} Deterministic lexical, rank-aware, and regex evaluators enable sub-minute pull-request gating at \$0.00 judge cost.
        \item \textbf{Future Directions:}
        \begin{enumerate}
            \item Multi-turn agentic RAG trajectory evaluation.
            \item Multimodal PDF table and chart grounding verification.
            \item Prediction-Powered Inference (PPI) calibration with lightweight NLI cross-encoders.
        \end{enumerate}
    \end{itemize}
\end{frame}
